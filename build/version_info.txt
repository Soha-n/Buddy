# Windows version resource for the frozen Python executables.
#
# Without this, right-click > Properties on buddy-backend.exe shows blank
# fields, which reads as unidentified software - and SmartScreen and antivirus
# heuristics both weigh missing metadata. The Tauri shell gets its own resource
# from tauri.conf.json; this covers the sidecars.
#
# filevers/prodvers must be 4-integer tuples, so 1.0.0 becomes (1, 0, 0, 0).
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          # 040904B0 = US English, Unicode.
          u'040904B0',
          [StringStruct(u'CompanyName', u'Buddy'),
           StringStruct(u'FileDescription', u'Buddy Service'),
           StringStruct(u'FileVersion', u'1.0.0.0'),
           StringStruct(u'InternalName', u'buddy-backend'),
           StringStruct(u'LegalCopyright', u'Copyright (C) 2026 Buddy'),
           StringStruct(u'OriginalFilename', u'buddy-backend.exe'),
           StringStruct(u'ProductName', u'Buddy'),
           StringStruct(u'ProductVersion', u'1.0.0.0')])
      ]),
    VarFileInfo([VarStruct(u'Translation', [0x0409, 1200])])
  ]
)
